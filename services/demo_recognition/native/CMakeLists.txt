cmake_minimum_required(VERSION 3.31)
project(foc_dlib_source_probe LANGUAGES CXX)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(PYBIND11_FINDPYTHON ON)
find_package(Python 3.13.13 EXACT REQUIRED COMPONENTS Interpreter Development.Module)
set(DLIB_NO_GUI_SUPPORT ON CACHE BOOL "" FORCE)
foreach(feature DLIB_USE_CUDA DLIB_USE_BLAS DLIB_USE_LAPACK DLIB_USE_MKL_FFT DLIB_USE_FFMPEG DLIB_LINK_WITH_SQLITE3 DLIB_PNG_SUPPORT DLIB_JPEG_SUPPORT DLIB_GIF_SUPPORT DLIB_WEBP_SUPPORT DLIB_JXL_SUPPORT USE_SSE4_INSTRUCTIONS USE_AVX_INSTRUCTIONS)
  set(${feature} OFF CACHE BOOL "" FORCE)
endforeach()
set(USE_SSE2_INSTRUCTIONS ON CACHE BOOL "" FORCE)
add_subdirectory(/build/source/dlib/external/pybind11 pybind11_build)
add_subdirectory(/build/source/dlib dlib_build)
pybind11_add_module(_foc_dlib_native NO_EXTRAS source_probe.cpp)
target_link_libraries(_foc_dlib_native PRIVATE dlib::dlib)
target_compile_features(_foc_dlib_native PRIVATE cxx_std_17)
